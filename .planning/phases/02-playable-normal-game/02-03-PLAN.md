Generated file format (Claude's discretion per CONTEXT; RESEARCH Pattern 1), one file per length at src/data/words-N.json:
  {"answers":"<space-joined sorted answers>","guesses":"<space-joined sorted guesses that are NOT answers>"}
  followed by a single "\n". The logical Guess Pool = answers + guesses. Plan 02-06's loader builds a Set from both strings.

scripts/words/lib.mjs (pure, no fs, no package imports):
- export function parseList(text: string): string[]           // strips '#' comments and blank lines, trims, lowercases, keeps file order
- export function isPlainWord(word: string): boolean           // /^[a-z]+$/
- export function hasVowel(word: string): boolean              // /[aeiouy]/
- export function isInflection(word: string, all: Set<string>): boolean   // D-02 stem-in-list rules, see Task 1
- export function buildPools({ length, common, all, slang, extra, blocklist, exclude, isProfane }) -> { answers: string[], guesses: string[], dropped: { inflection: string[], profanity: string[], blocklist: string[], exclude: string[], slang: string[], noVowel: string[] } }
    common/all/blocklist/exclude: Set<string>; slang: string[] (words); extra: string[]; isProfane: (w: string) => boolean
    answers = common words of `length` that are plain, hasVowel, not isInflection(all), not isProfane, not in blocklist, not in exclude, not in slang; sorted
    guesses = sorted unique (all words of `length` that are plain) + answers + slang of `length` + extra of `length`, MINUS answers (non-answer guesses only)
- export function assertInvariants({ length, answers, guesses, slang, blocklist, exclude, minAnswers }) -> void   // throws Error with a clear message on any violation
- export function serialisePools({ answers, guesses }) -> string    // JSON.stringify({ answers: answers.join(' '), guesses: guesses.join(' ') }) + '\n'
- export function renderSlangReview(entries, { isProfane, blocklist, commonByLength }) -> string   // markdown, see Task 2
